# Word Pattern

```cpp

class Solution {
public:
    bool wordPattern(string pattern, string s) {
        int n = pattern.length();

        vector<string> words;
        int countWords = 0;

        // amake srtingstream use korte hobe
        stringstream ss(s);
        string token;

        while (getline(ss, token, ' ')) {

            words.push_back(token);
            countWords++;
        }

        // first condition
        if (n != countWords)
            return false;

        // ekta hashmap use korte hobe & akta set use korte hobe
        unordered_map<string, char> mp;
        unordered_set<char> usedChar;

        for (int i = 0; i < n; i++) {

            string word = words[i];
            char ch = pattern[i];

            // check age map e ache kina and set e ache kina
            if (!mp.count(word) && !usedChar.count(ch)) {
                usedChar.insert(ch);
                mp[word] = ch;
            } else if (mp[word] != ch)
                return false;
        }

        return true;
    }
};
```
