# 2336. Smallest Number in Infinite Set

```cpp

class SmallestInfiniteSet {
public:

    vector<bool> nums;
    int i;

    SmallestInfiniteSet() {
        nums = vector<bool> (1001, true);
        i = 1; // 1 holo current smallest
    }
    
    int popSmallest() {
        int result = i;
        nums[i] = false;

        // i++ kora jabe na 
        for(int j=i+1;j<1001;j++) {
            if(nums[j] == true) {
                i = j;
                break;
            }
        }

        return result;
    }
    
    void addBack(int num) {
        // mane amake nums array te true korte hobe element take
        nums[num] = true;

        // jodi eta current smallest er thke choto hoi tahole i oke point korbe
        if(num < i) {
            i = num;
        }
    }
};

/**
 * Your SmallestInfiniteSet object will be instantiated and called as such:
 * SmallestInfiniteSet* obj = new SmallestInfiniteSet();
 * int param_1 = obj->popSmallest();
 * obj->addBack(num);
```
