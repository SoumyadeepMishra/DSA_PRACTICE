# 846. Hand of Straights

```cpp

class Solution {
public:
    bool isNStraightHand(vector<int>& hand, int groupSize) {
        int n = hand.size();
        // ordered map e store korbo
        map<int, int> mp;

        // map e element bhoro
        for(int &el : hand) {
            mp[el]++;
        }

        // take into the map
        while(!mp.empty()) {
            int curr = mp.begin()->first; // map er first element
            
            for(int i=0;i<groupSize;i++) {
                if(mp[curr + i] == 0)
                    return false;

                mp[curr+i]--;
                if(mp[curr+i] < 1)
                    mp.erase(curr+i);
            }

        }

        return true;
    }
};
```
