# 823. Binary Trees With Factors

```cpp

class Solution {
public:

    int mod = 1e9 + 7;

    int numFactoredBinaryTrees(vector<int>& arr) {
        int n = arr.size();
        unordered_map<int, long long> mp;
        int result = 0;

        sort(arr.begin(), arr.end());

        // sob element gulo ke nite hobe 
        for(int i=0;i<n;i++) {
            
            int root = arr[i];
            mp[root] = 1;

            for(int j=0;j<i;j++) {

                int lc = arr[j];
                int rc = root / lc;
                if(root % lc == 0 && mp.count(rc)) {
                    mp[root] += mp[lc] * mp[rc];
                }

            }
        }

        // traverse through the map
        for(auto &it : mp) {
            result = (result + it.second) % mod;
        }

        return result;
    }
};
```
