# 1743. Restore the Array From Adjacent Pairs

```cpp

class Solution {
public:

    vector<int> result;

    void dfs(int u, int prev, unordered_map<int, vector<int>> &adj) {
        result.push_back(u);

        for(int &v : adj[u]) {
            if(v != prev) {
                dfs(v, u, adj);
            }
        }

        return;
    }

    vector<int> restoreArray(vector<vector<int>>& adjacentPairs) {
        // prothome adjacency list banabo
        unordered_map<int, vector<int>> adj;
        for(vector<int> &vec : adjacentPairs) {
            int u = vec[0];
            int v = vec[1];

            adj[u].push_back(v);
            adj[v].push_back(u);
        }

        // starting point select korbo
        int startingPoint = -1;
        for(auto &it : adj) {
            if(it.second.size() == 1) {
                startingPoint = it.first;
            }
        }

        dfs(startingPoint, INT_MIN, adj);

        return result;

    }
};
```
